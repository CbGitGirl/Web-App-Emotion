import importlib.util
from pathlib import Path
from types import SimpleNamespace

for filename in ("emotion_detection.py", "2a_emotion_detection.py"):
    path = Path(filename)
    spec = importlib.util.spec_from_file_location(filename[:-3], path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)

    scores = {
        "anger": 0.01,
        "disgust": 0.02,
        "fear": 0.03,
        "joy": 0.91,
        "sadness": 0.04,
    }

    class Model:
        def run(self, text):
            return {"emotion_prediction": {"emotion": scores}}

    module.watson_nlp = SimpleNamespace(load=lambda name: Model())
    assert module.emotion_detector("hello")["dominant_emotion"] == "joy"
    assert module.emotion_detector(" ")["dominant_emotion"] is None

print("both Watson-compatible detector copies passed smoke checks")
